casesExact = categories(exactPatient.CaseNo);
looExact = table;
for i = 1:numel(casesExact)
excludedCase = casesExact{i};
fprintf('Exact LOO: excluding %s (%d/%d)\n', ...
excludedCase,i,numel(casesExact));
T = exactPatient( ...
exactPatient.CaseNo ~= excludedCase,:);
m = fitlme( ...
T, ...
'MeanWTC ~ State*ConnectionType + (1|CaseNo)');
A = anova(m,'DFMethod','satterthwaite');
intRow = strcmp(A.Term,'State:ConnectionType');
C = m.Coefficients;
antName  = ...
'State_Anterior:ConnectionType_Inter';
postrName = ...
'State_Posterior:ConnectionType_Inter';
postName = ...
'State_Post:ConnectionType_Inter';
antRow   = strcmp(C.Name,antName);
postrRow = strcmp(C.Name,postrName);
postRow  = strcmp(C.Name,postName);
R = table( ...
string(excludedCase), ...
A.FStat(intRow), ...
A.DF1(intRow), ...
A.DF2(intRow), ...
A.pValue(intRow), ...
C.Estimate(antRow), ...
C.pValue(antRow), ...
C.Estimate(postrRow), ...
C.pValue(postrRow), ...
C.Estimate(postRow), ...
C.pValue(postRow), ...
'VariableNames', ...
{'ExcludedCase', ...
'Interaction_F', ...
'Interaction_DF1', ...
'Interaction_DF2', ...
'Interaction_P', ...
'AnteriorInter_Beta', ...
'AnteriorInter_P', ...
'PosteriorInter_Beta', ...
'PosteriorInter_P', ...
'PostInter_Beta', ...
'PostInter_P'});
looExact = [looExact;R];
end
