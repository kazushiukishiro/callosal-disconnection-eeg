looPrimary = table;
looRegion  = table;
%% =========================================================
% 2. Leave-one-out loop
% ==========================================================
for i = 1:numel(cases)
excludedCase = cases{i};
fprintf('\nExcluding case %s (%d/%d)\n', ...
excludedCase,i,numel(cases));
T = wtc32(wtc32.CaseNo ~= excludedCase,:);
%% -----------------------------------------------------
% Primary WTC
% -----------------------------------------------------
mPrimary = fitlme( ...
T, ...
['mean_MeanWTC_4_13 ~ State + Region + ' ...
'(1|CaseNo) + (1|CaseNo:Region)']);
A = anova(mPrimary,'DFMethod','satterthwaite');
stateRow = strcmp(A.Term,'State');
C = mPrimary.Coefficients;
antRow  = strcmp(C.Name,'State_Anterior');
postrRow = strcmp(C.Name,'State_Posterior');
postRow = strcmp(C.Name,'State_Post');
R = table( ...
string(excludedCase), ...
A.FStat(stateRow), ...
A.DF1(stateRow), ...
A.DF2(stateRow), ...
A.pValue(stateRow), ...
C.Estimate(antRow), ...
C.pValue(antRow), ...
C.Estimate(postrRow), ...
C.pValue(postrRow), ...
C.Estimate(postRow), ...
C.pValue(postRow), ...
'VariableNames', ...
{'ExcludedCase', ...
'State_F','State_DF1','State_DF2','State_P', ...
'Anterior_Beta','Anterior_P', ...
'Posterior_Beta','Posterior_P', ...
'Post_Beta','Post_P'});
looPrimary = [looPrimary; R];
%% -----------------------------------------------------
% State x Region
% -----------------------------------------------------
mRegion = fitlme( ...
T, ...
['mean_MeanWTC_4_13 ~ State*Region + ' ...
'(1|CaseNo) + (1|CaseNo:Region)']);
AR = anova(mRegion,'DFMethod','satterthwaite');
intRow = strcmp(AR.Term,'State:Region');
R2 = table( ...
string(excludedCase), ...
AR.FStat(intRow), ...
AR.DF1(intRow), ...
AR.DF2(intRow), ...
AR.pValue(intRow), ...
'VariableNames', ...
{'ExcludedCase','Interaction_F', ...
'Interaction_DF1','Interaction_DF2', ...
'Interaction_P'});
looRegion = [looRegion; R2];
end
